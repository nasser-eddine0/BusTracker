import { useEffect, useState } from "react";
import { onValue, ref } from "firebase/database";
import { db } from "../firebase";
import { useAuth } from "../context/AuthContextObject";

export default function useFirebaseActiveTrip(tripId) {
  const { schoolId } = useAuth();
  const [activeTrip, setActiveTrip] = useState(null);
  const effectiveTripId = tripId || null;

  useEffect(() => {
    if (!schoolId || !effectiveTripId) {
      return undefined;
    }

    const unsubscribe = onValue(ref(db, `active_trips/${schoolId}/${effectiveTripId}`), (snapshot) => {
      setActiveTrip(snapshot.val() || null);
    });

    return () => unsubscribe();
  }, [effectiveTripId, schoolId]);

  return schoolId && effectiveTripId ? activeTrip : null;
}

import { useEffect, useState } from "react";
import { onValue, ref } from "firebase/database";
import { db } from "../firebase";
import { useAuth } from "../context/AuthContextObject";

/**
 * Listens to Firebase active_trips and returns the trip ID for a given bus.
 * This lets the parent detect in real-time when a driver starts/ends a trip
 * without needing to refresh the page.
 */
export default function useFirebaseTripForBus(busId) {
  const { schoolId } = useAuth();
  const [tripId, setTripId] = useState(null);
  const effectiveBusId = busId ? String(busId) : null;

  useEffect(() => {
    if (!schoolId || !effectiveBusId) {
      return undefined;
    }

    const unsubscribe = onValue(ref(db, `active_trips/${schoolId}`), (snapshot) => {
      const activeTrips = snapshot.val() || {};
      const matchingEntry = Object.entries(activeTrips).find(
        ([, trip]) => String(trip?.bus_id) === effectiveBusId && trip?.status === "in_progress"
      );
      setTripId(matchingEntry ? String(matchingEntry[0]) : null);
    });

    return () => unsubscribe();
  }, [effectiveBusId, schoolId]);

  return schoolId && effectiveBusId ? tripId : null;
}

          title: t("busVeryClose"),
          message: t("busWithin400m"),

          return [nextNotification, ...current].slice(0, 50);

    }

        const nextNotification = {
          id: `proximity-close-${studentWithLiveStatus.id}-trip-${bus.activeTripId}`,
          title: t("busApproaching"),
          message: "The bus is arriving soon!",
          studentName: studentWithLiveStatus.name,

        setNotifications((current) => {

          return [nextNotification, ...current].slice(0, 50);

      window.sessionStorage.removeItem(proximityStorageKey);
    }
  }, [bus?.activeTripId, busWithLiveLocation?.location, distanceToBus, homeLocation, studentWithLiveStatus, t]);

        await update(ref(db, `active_trips/${schoolId}/${effectiveTripId}/live_attendance/${studentWithLiveStatus.id}`), {
          status: "ready",
          updated_at: new Date().toISOString(),
        });
      } catch { /* best-effort */ }
    }

      await declareParentReady(studentWithLiveStatus.id);
    } catch { /* notification is best-effort */ }
  };

    setShowAbsenceModal(true);
  }, [readyState, savingAbsence, studentWithLiveStatus?.id]);

  const handleCloseAbsenceModal = useCallback(() => {
    if (savingAbsence) return;
    setShowAbsenceModal(false);
  }, [savingAbsence]);

  const handleNotComing = async () => {
    if (!studentWithLiveStatus?.id) return;

    setSavingAbsence(true);
    try {
      const response = await declareParentAbsence(studentWithLiveStatus.id);
      setStudent(response.data.student);

      // Update Firebase so the driver sees "absent" status in real-time
      if (schoolId && effectiveTripId) {
        try {
          await update(ref(db, `active_trips/${schoolId}/${effectiveTripId}/live_attendance/${studentWithLiveStatus.id}`), {
            status: "absent",
            updated_at: new Date().toISOString(),
          });
        } catch {
          // Absence is already persisted in the backend; ignore realtime mirror failures.
        }
      }

      setReadyState("not-coming");
      setShowAbsenceModal(false);
      toast.success(t("absenceConfirmed"));
    } catch (error) {
      toast.error(error?.response?.data?.message || t("absenceError"));
    } finally {
      setSavingAbsence(false);
    }
  };

  const handleLogout = async () => {
    await signOut();
    navigate("/signin", { replace: true });
  };

  const etaMinutes = distanceToBus === null ? null : Math.max(1, Math.round(distanceToBus * 4));

  const renderTabContent = () => {
    switch (activeTab) {
      case "map":
        return (
          <MapTab
            t={t}
            busWithLiveLocation={busWithLiveLocation}
            student={studentWithLiveStatus}
            distanceToBus={distanceToBus}
            etaMinutes={etaMinutes}
            proximityAlert={proximityAlert}
          />
        );
      case "alerts":
        return <AlertsTab t={t} unreadCount={unreadCount} notificationFeed={notificationFeed} onMarkRead={handleMarkParentRead} />;
      case "profile":
        return <ProfileTab t={t} user={user} student={studentWithLiveStatus} busWithLiveLocation={busWithLiveLocation} statusConfig={statusConfig} handleLogout={handleLogout} />;
      default:
        return (
          <HomeTab
            t={t}
            statusConfig={statusConfig}
            distanceToBus={distanceToBus}
            busWithLiveLocation={busWithLiveLocation}
            etaMinutes={etaMinutes}
            student={studentWithLiveStatus}
            assignmentMessage={assignmentMessage}
            proximityAlert={proximityAlert}
            readyState={readyState}
            savingAbsence={savingAbsence}
            notifications={notifications}
            notificationFeed={notificationFeed}
            handleReady={handleReady}
            handleOpenAbsenceModal={handleOpenAbsenceModal}
            setActiveTab={setActiveTab}
          />
        );
    }

  return (

              <img src="/logo.png" alt="BusTracker logo" className="h-8 w-8 object-contain" />

          <MotionDiv
            key={activeTab}
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -8 }}
            transition={{ duration: 0.22, ease: "easeOut" }}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;

class AuthController extends Controller

    public function register(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'email', 'max:255', 'unique:users,email'],
            'password' => ['required', 'string', 'min:8'],
            'phone' => ['nullable', 'string', 'max:40'],
            'role' => ['nullable', Rule::in(['parent'])],
        ]);

        $user = User::create([
            'name' => $validated['name'],
            'email' => strtolower($validated['email']),
            'password' => $validated['password'],
            'phone' => $validated['phone'] ?? null,
            'role' => 'parent',
            'status' => 'active',
        ]);

        $token = $user->createToken('auth_token')->plainTextToken;

        return response()->json([
            'access_token' => $token,
            'token_type' => 'Bearer',
            'user' => $this->serializeUser($user),
        ], 201);
    }

    public function login(Request $request, AuditLogger $auditLogger): JsonResponse
    {
        $validated = $request->validate([
            'email' => ['required', 'email'],
            'password' => ['required', 'string'],
            'role' => ['nullable', Rule::in(['parent', 'driver', 'admin', 'super_admin'])],
        ]);

        $user = User::where('email', strtolower($validated['email']))->first();

        if (!$user || !Hash::check($validated['password'], $user->password)) {
            return response()->json([
                'message' => 'Incorrect email or password.',
            ], 401);
        }

        if (($validated['role'] ?? null) && $user->role !== $validated['role']) {
            return response()->json([
                'message' => 'This account does not match the selected role.',
            ], 403);
        }

        if ($user->status !== 'active') {
            return response()->json([
                'message' => 'This account is disabled.',
            ], 403);
        }

        $user->tokens()->delete();
        $token = $user->createToken('auth_token')->plainTextToken;

        $auditLogger->log([
            'action' => 'user_logged_in',
            'description' => "User '{$user->email}' logged in",
            'user_id' => $user->id,
            'school_id' => $user->school_id,
            'new_values' => [
                'role' => $user->role,
                'email' => $user->email,
            ],
        ]);

        return response()->json([
            'access_token' => $token,
            'token_type' => 'Bearer',
            'user' => $this->serializeUser($user),
        ]);
    }

    public function me(Request $request): JsonResponse
    {
        return response()->json([
            'user' => $this->serializeUser($request->user()),
        ]);
    }

    public function logout(Request $request): JsonResponse
    {
        $request->user()?->currentAccessToken()?->delete();

        return response()->json([
            'message' => 'Logged out.',
        ]);
    }

    public function firebaseToken(Request $request): JsonResponse
    {
        $user = $request->user();
        
        if (!$user) {
            return response()->json([
                'message' => 'Unauthenticated.',
            ], 401);
        }

        try {
            $firebaseToken = app(\App\Services\FirebaseService::class)->createCustomTokenForUser($user->id);

            return response()->json([
                'firebaseToken' => $firebaseToken,
            ]);
        } catch (\App\Exceptions\FirebaseRequestException $e) {
            return response()->json([
                'message' => 'Failed to generate Firebase token.',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

    private function serializeUser(User $user): array
    {
        $user->loadMissing(['drivenBus', 'children']);

        return [
            'id' => (string) $user->id,
            'name' => $user->name,
            'email' => $user->email,
            'phone' => $user->phone,
            'role' => $user->role,
            'status' => $user->status,
            'cin' => $user->cin,
            'schoolId' => $user->school_id ? (string) $user->school_id : null,
            'busId' => $user->drivenBus?->id ? (string) $user->drivenBus->id : null,
            'studentId' => $user->children->count() === 1 ? (string) $user->children->first()->id : null,
        ];
    }
}

        ]);

            ->where('reg_code', $regCode)
            ->first();

                'message' => 'The selected registration code is invalid.',

                'message' => 'Your school subscription is inactive or expired. Please update billing to continue.',
            ], 403);

                    'reg_code' => ['This child is already linked to another parent account.'],
                ],

        }

        DB::transaction(function () use ($parent, $student, $firebaseService) {
            if ($parent->school_id === null) {

                // Update Firebase school_users mapping for the parent
                $firebaseService->createCustomTokenForUser($parent->id);
            }
            $student->update(['parent_id' => $parent->id]);
        });

        return response()->json([

            'student' => $this->serializeStudent($student->fresh(['bus', 'parent'])),

                'payload' => [
                    'studentName' => $student->full_name,
                    'parentName' => $parent->name,
                ],
            ]);

            $broadcaster->broadcast($notification);
        }

        $notificationSchoolId = (int) $student->school_id;

        foreach ($adminRecipients->activeForSchool($notificationSchoolId) as $admin) {
            $notification = Notification::create([
                'school_id' => $notificationSchoolId,
                'recipient_user_id' => $admin->id,
                'parent_id' => $parent->id,
                'student_id' => $student->id,
                'bus_id' => $student->bus_id,

        }

            'student' => $this->serializeStudent($student->fresh(['bus', 'parent'])),
        ]);
    }

    public function notifications(Request $request): JsonResponse
    {
        $schoolId = (int) $request->user()->school_id;

        return response()->json([
            'notifications' => $this->serializeNotifications($request->user()->id, $schoolId),
        ]);
    }

    public function markNotificationsRead(Request $request): JsonResponse
    {
        $schoolId = (int) $request->user()->school_id;

        Notification::query()
            ->where('school_id', $schoolId)
            ->where('recipient_user_id', $request->user()->id)
            ->whereNull('read_at')
            ->update(['read_at' => now()]);

        return response()->json(['message' => 'Notifications marked as read.']);
    }

    private function serializeNotifications(int $userId, int $schoolId): array
    {
        return Notification::query()
            ->where('school_id', $schoolId)
            ->where('recipient_user_id', $userId)
            ->latest()
            ->take(50)
            ->get()
            ->map(fn (Notification $notification) => [
                'id' => (string) $notification->id,
                'studentId' => $notification->student_id ? (string) $notification->student_id : null,
                'type' => $notification->type,
                'title' => $notification->title,
                'message' => $notification->message,
                'studentName' => $notification->payload['studentName'] ?? null,
                'payload' => $notification->payload,
                'read' => $notification->read_at !== null,
                'createdAt' => optional($notification->date_envoi ?? $notification->created_at)->toIso8601String(),
            ])
            ->all();
    }

    private function serializeStudent(Student $student): array
    {
        return [
            'id' => (string) $student->id,
            'name' => $student->full_name,
            'grade' => $student->grade,
            'address' => $student->address,
            'parentId' => $student->parent_id ? (string) $student->parent_id : '',
            'parentName' => $student->parent?->name ?? '',
            'parentEmail' => $student->parent?->email ?? '',
            'parentPhone' => $student->parent?->phone ?? '',
            'busId' => $student->bus_id ? (string) $student->bus_id : '',
            'status' => $student->status,
            'locationConformee' => (bool) $student->location_conformee,
            'homeLocation' => ($student->latitude !== null && $student->longitude !== null)
                ? ['lat' => (float) $student->latitude, 'lng' => (float) $student->longitude]
                : null,
        ];
    }

    private function serializeBus(Bus $bus): array
    {
        $latestTrip = $bus->trips()->latest('trip_date')->latest('id')->first();

        return [
            'id' => (string) $bus->id,
            'name' => $bus->bus_name,
            'routeName' => $bus->route_name,
            'driverName' => $bus->driver?->name ?? '',
            'plateNumber' => $bus->plate_number,
            'tripStatus' => $bus->trip_status,
            'activeTripId' => $bus->activeTrip?->id ? (string) $bus->activeTrip->id : null,
            'latestTripType' => $latestTrip?->type,
            'location' => ($bus->latitude_actuelle !== null && $bus->longitude_actuelle !== null)
                ? ['lat' => (float) $bus->latitude_actuelle, 'lng' => (float) $bus->longitude_actuelle]
                : null,
        ];
    }

    private function tenantExists(string $table, int $schoolId, string $column = 'id'): \Illuminate\Validation\Rules\Exists
    {
        return Rule::exists($table, $column)->where(fn ($query) => $query->where('school_id', $schoolId));
    }

<?php

namespace App\Services;

use App\Exceptions\FirebaseRequestException;
use Kreait\Firebase\Contract\Database;
use Kreait\Firebase\Factory;
use Kreait\Firebase\Auth as FirebaseAuth;
use Throwable;

class FirebaseService
{
    private Database $database;
    private FirebaseAuth $firebaseAuth;

    public function __construct()
    {
        $credentials = config('services.firebase.credentials');
        $databaseUrl = config('services.firebase.database_url');

        if (!$credentials || !$databaseUrl) {
            throw new FirebaseRequestException('Firebase credentials and database URL must be configured.');
        }

        $credentialsPath = base_path($credentials);
        if (!is_file($credentialsPath)) {
            $credentialsPath = $credentials;
        }

        $factory = (new Factory())
            ->withServiceAccount($credentialsPath)
            ->withDatabaseUri($databaseUrl);

        $this->database = $factory->createDatabase();
        $this->firebaseAuth = $factory->createAuth();
    }

    public function updateBusLocation(int $busId, array $data): void
    {
        $schoolId = $this->resolveSchoolId($data);
        $this->runWithRetry(fn () => $this->database->getReference("active_trips/{$schoolId}/{$busId}")->update($data));
    }

    public function updateStudentStatus(int $studentId, string $status): void
    {
        $schoolId = $this->resolveSchoolId();
        $this->runWithRetry(fn () => $this->database->getReference("students/{$schoolId}/{$studentId}")->update([
            'status' => $status,
        ]));
    }

    public function completeTrip(int $busId): void
    {
        $schoolId = $this->resolveSchoolId();
        $this->runWithRetry(fn () => $this->database->getReference("active_trips/{$schoolId}/{$busId}")->remove());
    }

    /**
     * Create a Firebase custom token for the given user ID and update school_users mapping.
     *
     * @param int $userId The Laravel user ID
     * @return string The Firebase custom token
     * @throws FirebaseRequestException
     */
    public function createCustomTokenForUser(int $userId): string
    {
        $user = \App\Models\User::find($userId);
        
        if (!$user) {
            throw new FirebaseRequestException('User not found.');
        }

        // Create Firebase custom token keyed by the Laravel user ID
        $token = $this->firebaseAuth->createCustomToken((string) $user->id);

        // Update school_users mapping so Firebase RTDB rules can validate the user
        // Only do this if the user belongs to a school (not super_admin)
        if ($user->school_id !== null) {
            $this->runWithRetry(fn () => $this->database->getReference("school_users/{$user->school_id}/{$user->id}")
                ->set([
                    'role' => $user->role,
                    'busId' => $user->drivenBus?->id ?? null,
                    'updatedAt' => now()->toISOString(),
                ]));
        }

        return $token;
    }

    private function runWithRetry(callable $operation): mixed
    {
        $lastError = null;

        for ($attempt = 1; $attempt <= 3; $attempt++) {
            try {
                return $operation();
            } catch (Throwable $exception) {
                $lastError = $exception->getMessage();

                if ($attempt < 3) {
                    usleep((int) (100000 * (2 ** ($attempt - 1))));
                }
            }
        }

        throw new FirebaseRequestException("Firebase Admin SDK request failed after 3 attempts: {$lastError}");
    }

    private function resolveSchoolId(array $data = []): int
    {
        $schoolId = $data['school_id'] ?? auth()->user()?->school_id;

        if (!$schoolId) {
            throw new FirebaseRequestException('Unable to resolve school_id for Firebase request.');
        }

        return (int) $schoolId;
    }
}

import { useCallback, useEffect, useMemo, useState } from "react";
import api from "../api/axios";
import { auth, signInWithCustomToken, signOut as firebaseSignOut } from "../firebase";
import { AuthContext } from "./AuthContextObject";
export { useAuth } from "./AuthContextObject";

function normalizeUser(user) {
  if (!user) return null;

  return {
    ...user,
    id: user.id ? String(user.id) : null,
    busId: user.busId ? String(user.busId) : null,
    studentId: user.studentId ? String(user.studentId) : null,
  };
}

function readCachedUser() {
  try {
    const rawUser = localStorage.getItem("auth_user");
    return rawUser ? normalizeUser(JSON.parse(rawUser)) : null;
  } catch {
    return null;
  }
}

function clearClientAuthState() {
  localStorage.removeItem("token");
  localStorage.removeItem("auth_user");

  try {
    sessionStorage.removeItem("admin_bootstrap_cache");
  } catch {
    // ignore storage cleanup failures
  }
}

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => readCachedUser());
  const [loading, setLoading] = useState(() => {
    if (typeof window === "undefined") return true;
    return Boolean(localStorage.getItem("token")) && !readCachedUser();
  });

    const refreshProfile = useCallback(async () => {
      const token = localStorage.getItem("token");
      if (!token) {
        setUser(null);
        setLoading(false);
        return null;
      }

      try {
        const response = await api.get("/me");
        const nextUser = normalizeUser(response.data.user);
        setUser(nextUser);
        localStorage.setItem("auth_user", JSON.stringify(nextUser));
        
        // Re-initialize Firebase authentication with fresh token
        try {
          const firebaseTokenResponse = await api.get("/firebase-token");
          await signInWithCustomToken(auth, firebaseTokenResponse.data.firebaseToken);
        } catch (error) {
          console.error("Failed to refresh Firebase authentication:", error);
          // Don't fail the profile refresh if Firebase auth fails
        }
        
        return nextUser;
      } catch {
        clearClientAuthState();
        setUser(null);
        return null;
      } finally {
        setLoading(false);
      }
    }, []);

  useEffect(() => {
    refreshProfile();
  }, [refreshProfile]);

    const signIn = useCallback(async (payload) => {
      const response = await api.post("/login", payload);
      localStorage.setItem("token", response.data.access_token);
      const nextUser = normalizeUser(response.data.user);
      localStorage.setItem("auth_user", JSON.stringify(nextUser));
      setUser(nextUser);
      setLoading(false);
      
      // Initialize Firebase authentication
      try {
        const firebaseTokenResponse = await api.get("/firebase-token");
        await signInWithCustomToken(auth, firebaseTokenResponse.data.firebaseToken);
      } catch (error) {
        console.error("Failed to initialize Firebase authentication:", error);
        // Don't fail the login if Firebase auth fails - the app can still work with REST API
      }
      
      return nextUser;
    }, []);

  const signUp = useCallback(async (payload) => {
    const response = await api.post("/register", payload);
    localStorage.setItem("token", response.data.access_token);
    const nextUser = normalizeUser(response.data.user);
    localStorage.setItem("auth_user", JSON.stringify(nextUser));
    setUser(nextUser);
    setLoading(false);
    return nextUser;
  }, []);

    const signOut = useCallback(async () => {
      try {
        await api.post("/logout");
      } catch {
        // local cleanup still matters even if the API call fails
      } finally {
        // Sign out from Firebase
        try {
          await firebaseSignOut(auth);
        } catch (error) {
          console.error("Failed to sign out from Firebase:", error);
        }
        
        clearClientAuthState();
        setUser(null);
        setLoading(false);
      }
    }, []);

  const value = useMemo(() => {
    return {
      user,
      profile: user,
      loading,
      role: user?.role || null,
      schoolId: user?.schoolId || null,
      busId: user?.busId || null,
      studentId: user?.studentId || null,
      signIn,
      signUp,
      signOut,
      refreshProfile,
    };
  }, [loading, refreshProfile, signIn, signOut, signUp, user]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

    goOnline(db);
  };

  const handleOffline = () => {
